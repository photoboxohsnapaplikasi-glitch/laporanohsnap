var SPREADSHEET_ID = '1YuG4fd5DzIbgye0i7DzTzugqtj_rDlap5-6a4qizLik';
var FOLDER_ABSEN_ID = '1YPsH4RmsrfPdqFZUN5A2vmszt_ZyKvUH';
var FOLDER_ABSEN_NAMA = 'Foto Absensi Open & Close Store - Oh Snap 5 Cabang';

function getFolderAbsen() {
  try {
    return DriveApp.getFolderById(FOLDER_ABSEN_ID);
  } catch (e) {
    var folders = DriveApp.getFoldersByName(FOLDER_ABSEN_NAMA);
    if (folders.hasNext()) return folders.next();
    return DriveApp.createFolder(FOLDER_ABSEN_NAMA);
  }
}

function tesTulisKeSpreadsheet() {
  var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  var folderAbsen = getFolderAbsen();
  var shAbsen = ss.getSheetByName('Absensi_Store') || ss.insertSheet('Absensi_Store');
  if (shAbsen.getLastRow() === 0) {
    shAbsen.appendRow([
      'Timestamp', 'Cabang', 'Tipe Absen', 'Tanggal', 'Jam', 'Shift', 'Nama Karyawan',
      'Koordinat / Watermark Lokasi', 'Ceklis / Keterangan',
      'Link Foto 1 (Muka Open / Lampu Mati Close)', 'Link Foto 2 (Depan Store Close)', 'Catatan'
    ]);
  }
  Logger.log('Berhasil! Folder Foto Absen: ' + folderAbsen.getUrl());
}

function doGet(e) {
  if (e && e.parameter && e.parameter.payload) {
    return ContentService.createTextOutput(simpanDariWebApp(e.parameter.payload))
      .setMimeType(ContentService.MimeType.JSON);
  }
  return ContentService.createTextOutput('Koneksi Spreadsheet & Folder Drive Oh Snap Photobox (5 Cabang) Aktif!');
}

function doPost(e) {
  var raw = '';
  if (e && e.parameter && e.parameter.payload) {
    raw = e.parameter.payload;
  } else if (e && e.postData && e.postData.contents) {
    raw = e.postData.contents;
  }
  return ContentService.createTextOutput(simpanDariWebApp(raw))
    .setMimeType(ContentService.MimeType.JSON);
}

function simpanDariWebApp(jsonString) {
  try {
    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    var d = JSON.parse(jsonString);
    var cabang = d.cabang || 'Cilacap';

    function saveImg(base64, name, isAbsenOpenClose) {
      try {
        if (!base64 || base64.indexOf('base64,') === -1) return '-';
        var targetFolder = isAbsenOpenClose ? getFolderAbsen() : DriveApp.getRootFolder();
        var parts = base64.split('base64,');
        var contentType = parts[0].replace('data:', '').replace(';', '');
        var blob = Utilities.newBlob(Utilities.base64Decode(parts[1]), contentType, name);
        return targetFolder.createFile(blob).getUrl();
      } catch (imgErr) {
        return 'Foto tersimpan di aplikasi (' + imgErr.message + ')';
      }
    }

    if (d.recordType === 'OPEN_STORE' || d.recordType === 'CLOSE_STORE') {
      var shAbsen = ss.getSheetByName('Absensi_Store') || ss.insertSheet('Absensi_Store');
      if (shAbsen.getLastRow() === 0) {
        shAbsen.appendRow([
          'Timestamp', 'Cabang', 'Tipe Absen', 'Tanggal', 'Jam', 'Shift', 'Nama Karyawan',
          'Koordinat / Watermark Lokasi', 'Ceklis / Keterangan',
          'Link Foto 1 (Muka Open / Lampu Mati Close)', 'Link Foto 2 (Depan Store Close)', 'Catatan'
        ]);
      }
      var tipeDetail = d.tipeAbsenDetail || d.recordType;
      if (d.recordType === 'OPEN_STORE') {
        var fotoOpenUrl = saveImg(d.fotoBukti, tipeDetail + '_' + cabang + '_' + d.tanggal + '_' + d.nama + '_' + d.shift + '.jpg', true);
        shAbsen.appendRow([
          new Date(), cabang, tipeDetail, d.tanggal, d.waktu || '-', d.shift, d.nama,
          d.lokasiWatermark || '-', d.checklistSummary || '-', fotoOpenUrl, '-', d.catatan || '-'
        ]);
      } else {
        var fotoLampuUrl = saveImg(d.fotoLampuMati || d.fotoBukti, 'CLOSE_LAMPU_MATI_' + cabang + '_' + d.tanggal + '_' + d.nama + '_' + d.shift + '.jpg', true);
        var fotoDepanUrl = saveImg(d.fotoDepanStore, 'CLOSE_DEPAN_STORE_' + cabang + '_' + d.tanggal + '_' + d.nama + '_' + d.shift + '.jpg', true);
        shAbsen.appendRow([
          new Date(), cabang, 'CLOSE_STORE', d.tanggal, d.waktu || '-', d.shift, d.nama,
          d.lokasiWatermark || '-', d.checklistSummary || '-', fotoLampuUrl, fotoDepanUrl, d.catatan || '-'
        ]);
      }
      return JSON.stringify({ status: 'success', folderUrl: 'https://drive.google.com/drive/folders/' + FOLDER_ABSEN_ID });
    }

    if (d.recordType === 'STOCK_OPNAME') {
      var shStok = ss.getSheetByName('Stock_Opname') || ss.insertSheet('Stock_Opname');
      if (shStok.getLastRow() === 0) {
        shStok.appendRow([
          'Timestamp', 'Cabang', 'Peran Pengirim', 'Tanggal Input', 'Tanggal Cutoff', 'Shift', 'Nama Petugas',
          'Kertas (Pax)', 'Hangtag Kecil & Besar (Pcs)', 'Keychain Panjang & Bulat (Pcs)',
          'Plastik Kecil & Besar (Ikat)', 'Tinta Printer (LM, LC, Y, M, C, K)', 'Lem Ganci (Pcs)',
          'Rincian Etalase Lengkap', 'Catatan'
        ]);
      }
      shStok.appendRow([
        new Date(), cabang, d.senderRole || 'USER', d.tanggal, d.cutoffDate || d.tanggal, d.shift, d.nama,
        d.kertasPax || '-', d.hangtagPcs || '-', d.keychainPcs || '-',
        d.plastikIkat || '-', d.tintaStatus || '-', d.lemGanciPcs || '-',
        d.itemsSummary || '-', d.catatan || '-'
      ]);
      return JSON.stringify({ status: 'success' });
    }

    return JSON.stringify({ status: 'success' });
  } catch (err) {
    return JSON.stringify({ status: 'error', message: err.toString() });
  }
}
