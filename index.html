var SPREADSHEET_ID = '1YuG4fd5DzIbgye0i7DzTzugqtj_rDlap5-6a4qizLik';
var FOLDER_ABSEN_ID = '1YPsH4RmsrfPdqFZUN5A2vmszt_ZyKvUH';
var FOLDER_ABSEN_NAMA = 'Foto Absensi Open & Close Store - Oh Snap Cilacap';

// Ambil folder khusus Foto Open & Close Store di Google Drive Anda
function getFolderAbsen() {
  try {
    return DriveApp.getFolderById(FOLDER_ABSEN_ID);
  } catch (e) {
    var folders = DriveApp.getFoldersByName(FOLDER_ABSEN_NAMA);
    if (folders.hasNext()) return folders.next();
    return DriveApp.createFolder(FOLDER_ABSEN_NAMA);
  }
}

// 1. Klik tombol "Run" (Jalankan) di atas untuk mengetes & memberi izin akses ke Spreadsheet + Google Drive Anda
function tesTulisKeSpreadsheet() {
  var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  var folderAbsen = getFolderAbsen();
  var sheet = ss.getSheets()[0];
  if (sheet.getLastRow() === 0) {
    sheet.appendRow([
      'Timestamp', 'Tanggal', 'Cabang', 'Shift', 'Nama Staf',

      'Total Pemasukan Harian (Bersih)', 'Link Foto Serah Terima (Wajib)', 'Link Foto Struk (Opsional)', 'Catatan'
    ]);
  }
  sheet.appendRow([
    new Date(), 'TES-KONEKSI', 'Cabang Cilacap', 'Shift Pagi (09.30 - 16.00)', 'Saskia',
    500000, 750000, 1250000,
    0, '-',
    1250000, folderAbsen.getUrl(), '-', 'Koneksi Spreadsheet & Folder Drive Absen (' + folderAbsen.getName() + ') berhasil aktif!'

}

function doGet(e) {
  if (e && e.parameter && e.parameter.payload) {
    return ContentService.createTextOutput(simpanDariWebApp(e.parameter.payload))
      .setMimeType(ContentService.MimeType.JSON);
  }
  return ContentService.createTextOutput('✅ Koneksi Spreadsheet & Folder Drive Oh Snap Photobox Cilacap Aktif!');
}

function doPost(e) {
  var raw = '';
  if (e && e.parameter && e.parameter.payload) {
    raw = e.parameter.payload;
  } else if (e && e.postData && e.postData.contents) {
    raw = e.postData.contents;
  }
  return ContentService.createTextOutput(simpanDariWebApp(raw))
    .setMimeType(ContentService.MimeType.JSON);
}

function simpanDariWebApp(jsonString) {
  try {
    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    var d = JSON.parse(jsonString);

    function saveImg(base64, name, isAbsenOpenClose) {
      try {
        if (!base64 || base64.indexOf('base64,') === -1) return '-';
        var targetFolder = isAbsenOpenClose ? getFolderAbsen() : DriveApp.getRootFolder();
        var parts = base64.split('base64,');
        var contentType = parts[0].replace('data:', '').replace(';', '');
        var blob = Utilities.newBlob(Utilities.base64Decode(parts[1]), contentType, name);
        return targetFolder.createFile(blob).getUrl();
      } catch (imgErr) {
        return 'Foto tersimpan di aplikasi (' + imgErr.message + ')';
      }
    }

    if (d.recordType === 'OPEN_STORE' || d.recordType === 'CLOSE_STORE') {
      var shAbsen = ss.getSheetByName('Absensi_Store') || ss.insertSheet('Absensi_Store');
      if (shAbsen.getLastRow() === 0) {
        shAbsen.appendRow([
          'Timestamp', 'Tipe Absen', 'Tanggal', 'Jam', 'Shift', 'Nama Petugas',
          'Koordinat / Watermark Lokasi', 'Ceklis Alat Photobooth',
          'Link Foto 1 (Selfie Open / Lampu Mati Close)', 'Link Foto 2 (Depan Store Close)', 'Catatan'
        ]);
      }
      if (d.recordType === 'OPEN_STORE') {
        var fotoOpenUrl = saveImg(d.fotoBukti, 'OPEN_SELFIE_' + d.tanggal + '_' + d.nama + '_' + d.shift + '.jpg', true);
        shAbsen.appendRow([
          new Date(), 'OPEN_STORE', d.tanggal, d.waktu || '-', d.shift, d.nama,
          d.lokasiWatermark || '-', d.checklistSummary, fotoOpenUrl, '-', d.catatan || '-'
        ]);
      } else {
        var fotoLampuUrl = saveImg(d.fotoLampuMati || d.fotoBukti, 'CLOSE_LAMPU_MATI_' + d.tanggal + '_' + d.nama + '_' + d.shift + '.jpg', true);
        var fotoDepanUrl = saveImg(d.fotoDepanStore, 'CLOSE_DEPAN_STORE_' + d.tanggal + '_' + d.nama + '_' + d.shift + '.jpg', true);
        shAbsen.appendRow([
          new Date(), 'CLOSE_STORE', d.tanggal, d.waktu || '-', d.shift, d.nama,
          d.lokasiWatermark || '-', d.checklistSummary, fotoLampuUrl, fotoDepanUrl, d.catatan || '-'
        ]);
      }
      return JSON.stringify({ status: 'success', folderUrl: 'https://drive.google.com/drive/folders/' + FOLDER_ABSEN_ID });
    }

    if (d.recordType === 'STOCK_OPNAME') {
      var shStok = ss.getSheetByName('Stock_Opname') || ss.insertSheet('Stock_Opname');
      if (shStok.getLastRow() === 0) {
        shStok.appendRow([
          'Timestamp', 'Tanggal Input', 'Tanggal Cutoff', 'Shift', 'Nama Petugas',
          'Kertas (Pax)', 'Hangtag Kecil & Besar (Pcs)', 'Keychain Panjang & Bulat (Pcs)',
          'Plastik Kecil & Bulat (Ikat)', 'Indikator Tinta', 'Lem Ganci (Pcs)',
          'Rincian Etalase Lengkap', 'Catatan'
        ]);
      }
      shStok.appendRow([
        new Date(), d.tanggal, d.cutoffDate || d.tanggal, d.shift, d.nama,
        d.kertasPax || '-', d.hangtagPcs || '-', d.keychainPcs || '-',
        d.plastikIkat || '-', d.tintaStatus || '-', d.lemGanciPcs || '-',
        d.itemsSummary || '-', d.catatan || '-'
      ]);
      return JSON.stringify({ status: 'success' });
    }

    var sheet = ss.getSheets()[0];
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        'Timestamp', 'Tanggal', 'Cabang', 'Shift', 'Nama Staf',
        'Pemasukan Cash', 'Pemasukan QRIS', 'Pemasukan Total Harian (Kotor)',
        'Total Pengeluaran', 'Rincian Pengeluaran',
        'Total Pemasukan Harian (Bersih)', 'Link Foto Serah Terima (Wajib)', 'Link Foto Struk (Opsional)', 'Catatan'
      ]);
    }

    var strukUrl = saveImg(d.fotoStruk, 'STRUK_' + d.tanggal + '_' + d.nama + '.jpg', false);
    var serahUrl = saveImg(d.fotoSerahTerima, 'SERAH_TERIMA_' + d.tanggal + '_' + d.nama + '.jpg', false);
    sheet.appendRow([
      new Date(), d.tanggal, d.cabang || 'Cabang Cilacap', d.shift, d.nama,
      d.cash || 0, d.qris || 0, d.totalGross || 0,
      d.totalExpense || 0, d.rincianPengeluaran || '-',
      d.netOmzet || 0, serahUrl, strukUrl, d.catatan || '-'
    ]);
    return JSON.stringify({ status: 'success' });
  } catch (err) {
    return JSON.stringify({ status: 'error', message: err.toString() });
  }
}
